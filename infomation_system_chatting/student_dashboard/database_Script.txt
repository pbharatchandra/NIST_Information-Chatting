-- 1. Create the table
CREATE TABLE timetable (
    id SERIAL PRIMARY KEY,
    day_name VARCHAR(20),       -- e.g., 'Monday'
    time_slot VARCHAR(50),      -- e.g., '09-10 AM'
    room_number VARCHAR(50),
    subject VARCHAR(50),
    type VARCHAR(20)            -- e.g., 'Lecture'
);
